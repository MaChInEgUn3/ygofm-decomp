# Extra names for addresses that splat gave a single symbol. See read_defsyms()
# in tools_src/build.py for why these are needed. Format matches splat's own
# undefined_syms files: NAME = 0xADDRESS;
#
# func_80036BCC walks a 2-byte key table and a 0x1E-byte record table that
# start at the same address, and materialises the base twice.
KeyTable_801D9174 = 0x801D9174;

# func_8002497C reads this counter once to test it and once to index with, in
# two different basic blocks, and materialises the address for each.
Base2_8009B364 = 0x8009B364;

# func_8003D0F4 reads this flag and writes it back in the same function with
# nothing between that could have changed it, so gcc commons the address.
Base2_8009B408 = 0x8009B408;

# func_800137E4 reads D_8009B0F4 three times and D_8009B134 twice across a
# wait loop; without extra names cc1psx hoists the addresses into callee-saved
# registers and grows the frame.
Base2_8009B0F4 = 0x8009B0F4;
Base3_8009B0F4 = 0x8009B0F4;
Base2_8009B134 = 0x8009B134;

# func_8003767C stores each of these twice; without a second name cc1psx
# commons the address into a callee-saved register and grows the frame.
Base2_8009B2AA = 0x8009B2AA;
# func_8003771C touches the same counter four times, so it needs three extra
# names -- one per access beyond the first.
Base3_8009B2AA = 0x8009B2AA;
Base4_8009B2AA = 0x8009B2AA;
Base2_8009B2A8 = 0x8009B2A8;


# splat found no symbol here; func_8002892C reaches it as 0x342($gp) and
# _gp is 0x8009AF08. Named so the C can refer to it.
D_8009B24A = 0x8009B24A;

# THREE of the five entries this file used to have were dead weight, measured
# rather than assumed: func_80044DA0 matches with both stores naming
# D_8009B45C, func_8002CCE4 sits at the same 3 differences either way, and
# func_80049C40 -- which looked like the textbook case, two pointer values
# into one object in one basic block -- matches with both naming D_8009B458.
# An alias is needed when the source needs two distinct *pointer values* into
# one object -- func_8002497C holds one in a variable while reading through the
# other, func_80036BCC walks two -- because gcc commons the address expression.
# It is NOT needed when the repeated %hi comes from separate bare-symbol memory
# ops, which the assembler expands one at a time; see func_8002A660 -- and it
# is not needed when a *store through one of the pointers* sits between the
# two reads, because D_8009B458 is itself a pointer in memory that the store
# might alias, so gcc has to reload it anyway. That is func_80049C40. What is
# left for the two surviving aliases is two reads of one address with nothing
# in between that could have changed it.

# func_80058E1C reads this limit twice, once to compare and once to return,
# with only the branch in between; without a second name gcc keeps the first
# load in a register and the second read disappears.
Base2_8009AFA3 = 0x8009AFA3;

# func_8003E46C stores the cleared value and then the or'd value to the same
# halfword; with one name gcc drops the first store.
Base2_8009B3FA = 0x8009B3FA;

# splat found no symbol here either; func_8004BBBC hoists %hi(0x8009AF80) into
# $s1 and hands the address to func_8004BAA0. It sits between D_8009AF7A and
# D_8009AF8E, so nothing else claims it.
D_8009AF80 = 0x8009AF80;

# Back for a second time, and this time measured: func_8004BBBC reads
# D_8009B458 twice inside a loop and retail re-materialises %hi for each,
# while gcc hoists one %hi into a callee-saved register and grows the frame by
# a register. The 2026-07-26 note below deleted this name because
# func_80049C40 did not need it -- that function has a store between the two
# reads, which forces the reload anyway. This one does not.
Base2_8009B458 = 0x8009B458;

# splat has D_8009AF2A and D_8009AF2D but not the byte between them, which
# func_80022618 passes as the second argument of a printf.
D_8009AF2C = 0x8009AF2C;

# splat has no symbol for the "bu%02X:%s" format string func_800443EC passes.
D_80010538 = 0x80010538;

# func_80012DB4 tests this flag and then clears it. Both accesses are the bare
# form, but gcc still commons the array base into a register when the same name
# appears twice, so the store needs its own name.
Base2_8009AFA4 = 0x8009AFA4;

# func_80047EC4 reads D_8009B45C twice inside a loop. Both are bare-symbol
# loads, but two references to one name make gcc form the address as a loop
# invariant and hoist it into a callee-saved register; two names leave each
# `lw symbol` alone, which is what retail has.
Base2_8009B45C = 0x8009B45C;

# func_8003D03C reads D_8009B408 signed, clears it, then reads it unsigned --
# three accesses in one function, so two extra names.
Base3_8009B408 = 0x8009B408;

# func_80045054 reaches D_8009B45C four times and retail materialises it four
# times; with one name gcc hoists the address into a callee-saved register and
# grows the frame. One name per reference beyond the first.
Base3_8009B45C = 0x8009B45C;
Base4_8009B45C = 0x8009B45C;
